## 队列

队列是一种特殊的线性表，只允许在表的前端 `front` 进行删除操作，在表的后端 `rear` 进行插入操作。

- 进行插入操作的端称为队尾。
- 进行删除操作的端称为队头。
- 队列中没有元素时称为空队列。
- 队列的特点：先进先出（FIFO）。

### 循环队列

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_SIZE 6

typedef struct
{
    int data[MAX_SIZE];
    int front;
    int rear;
} QUEUE;

void En_Queue(QUEUE *q);
void De_Queue(QUEUE *q);

int main()
{
    QUEUE q = {0};
    int choose;

    while (1)
    {
        printf("**********1.入队**********\n");
        printf("**********2.出队**********\n");
        printf("**********0.退出**********\n");
        scanf("%d", &choose);

        switch (choose)
        {
        case 1:
            En_Queue(&q);
            break;
        case 2:
            De_Queue(&q);
            break;
        case 0:
            return 0;
        }
    }

    return 0;
}

void En_Queue(QUEUE *q)
{
    if ((q->rear + 1) % MAX_SIZE == q->front)
    {
        printf("队满,无法入队\n");
        return;
    }

    printf("请输入入队数据:\n");
    scanf("%d", &q->data[q->rear]);
    q->rear = (q->rear + 1) % MAX_SIZE;
    printf("入队成功\n");
}

void De_Queue(QUEUE *q)
{
    if (q->front == q->rear)
    {
        printf("队空,无法出队\n");
        return;
    }

    printf("出队数据:%d\n", q->data[q->front]);
    q->front = (q->front + 1) % MAX_SIZE;
    printf("出队成功\n");
}
```

队满判断：

```c
(q->rear + 1) % MAX_SIZE == q->front
```

队空判断：

```c
q->front == q->rear
```

### 链式队列

链式队列使用链表实现。

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

typedef struct Queue
{
    int data;//数据域
    struct Queue *next;
} QUEUE;

void En_Queue(QUEUE **front, QUEUE **rear);
void De_Queue(QUEUE **front, QUEUE **rear);

int main()
{
    QUEUE *front = NULL;
    QUEUE *rear = NULL;
    int choose;

    while (1)
    {
        printf("**********1.入队**********\n");
        printf("**********2.出队**********\n");
        printf("**********0.退出**********\n");
        scanf("%d", &choose);

        switch (choose)
        {
        case 1:
            En_Queue(&front, &rear);
            break;
        case 2:
            De_Queue(&front, &rear);
            break;
        case 0:
            return 0;
        }
    }

    return 0;
}

/**********************************
函数的功能：入队函数
函数的返回值：无
函数的参数：
***********************************/
void En_Queue(QUEUE **front, QUEUE **rear)
{
    QUEUE *new = (QUEUE *)malloc(sizeof(QUEUE));

    if (new == NULL)
    {
        perror("malloc");
        return;
    }

    memset(new, 0, sizeof(QUEUE));
    //数据域和指针域赋值
    //开始入队
    printf("请输入入队数据:\n");
    scanf("%d", &new->data);
    new->next = NULL;

    if (*rear == NULL)
    {
        *front = *rear = new;
    }
    else
    {
        (*rear)->next = new;
        *rear = new;
    }

    printf("入队成功\n");
}

/**********************************
函数的功能：出队函数
函数的返回值：无
函数的参数：
***********************************/
void De_Queue(QUEUE **front, QUEUE **rear)
{
    //先判断是否队空
    if (*front == NULL)
    {
        printf("队空,无法出队\n");
        return;
    }

    printf("出队数据:%d\n", (*front)->data);
    //删除队头节点
    QUEUE *del = *front;
    *front = (*front)->next;
    free(del);

    if (*front == NULL)
    {
        *rear = NULL;
    }

    printf("出队成功\n");
}
```
