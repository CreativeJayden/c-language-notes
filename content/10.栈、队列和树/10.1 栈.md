## 栈

堆栈又名栈，是一种运算受限的线性表。只在表尾进行插入和删除操作，这一端称为栈顶，另一端称为栈底。

向栈插入新元素称为进栈、入栈或压栈；从栈删除元素称为出栈或退栈。

栈的特点：先进后出（FILO）。

实例：弹夹、糖葫芦等。

### 顺序栈

顺序栈使用数组实现。

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_SIZE 6

typedef struct
{
    int data[MAX_SIZE];
    int top;
} STACK;

void Push_Stack(STACK *sta);
void Pop_Stack(STACK *sta);

int main()
{
    STACK sta = {.top = -1};
    int choose;

    while (1)
    {
        printf("**********1.入栈**********\n");
        printf("**********2.出栈**********\n");
        printf("**********0.退出**********\n");
        scanf("%d", &choose);

        switch (choose)
        {
        case 1:
            Push_Stack(&sta);
            break;
        case 2:
            Pop_Stack(&sta);
            break;
        case 0:
            return 0;
        }
    }

    return 0;
}

void Push_Stack(STACK *sta)
{
    printf("----------->入栈\n");

    if (sta->top == MAX_SIZE - 1)
    {
        printf("栈满,无法入栈\n");
        return;
    }

    printf("请输入入栈数据\n");
    sta->top++;
    scanf("%d", &sta->data[sta->top]);
    printf("入栈成功\n");
}

void Pop_Stack(STACK *sta)
{
    printf("----------->出栈\n");

    if (sta->top == -1)
    {
        printf("栈空,无法出栈\n");
        return;
    }

    printf("出栈数据:%d\n", sta->data[sta->top]);
    sta->top--;
    printf("出栈成功\n");
}
```

栈空判断：

```c
sta->top == -1
```

栈满判断：

```c
sta->top == MAX_SIZE - 1
```

### 链式栈

链式栈使用链表实现。

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

typedef struct Stack
{
    int data;
    struct Stack *front;
} STACK;

void Push_Stack();
void Pop_Stack();

STACK *top = NULL;

int main()
{
    int choose;

    while (1)
    {
        printf("**********1.入栈**********\n");
        printf("**********2.出栈**********\n");
        printf("**********0.退出**********\n");
        scanf("%d", &choose);

        switch (choose)
        {
        case 1:
            Push_Stack();
            break;
        case 2:
            Pop_Stack();
            break;
        case 0:
            return 0;
        }
    }

    return 0;
}

/**********************************
函数的功能：入栈函数
函数的返回值：无
函数的参数：无
***********************************/
void Push_Stack()
{
    printf("----------->入栈\n");
    STACK *new = (STACK *)malloc(sizeof(STACK));

    if (new == NULL)
    {
        perror("malloc");
        return;
    }

    //开始入栈
    printf("请输入入栈数据\n");
    scanf("%d", &new->data);
    new->front = top;
    top = new;
    printf("入栈成功\n");
}

/**********************************
函数的功能：出栈函数
函数的返回值：无
函数的参数：无
***********************************/
void Pop_Stack()
{
    printf("----------->出栈\n");

    //先判断是否栈空
    if (top == NULL)
    {
        printf("栈空,无法出栈\n");
        return;
    }

    //出栈
    printf("出栈数据:%d\n", top->data);
    STACK *del = top;
    top = top->front;
    free(del);
    printf("出栈成功\n");
}
```
